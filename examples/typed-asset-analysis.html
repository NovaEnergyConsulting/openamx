Typed Asset Screening Analysis

This illustrative screening combines a nested Asset Management record from JSON with scalar screening rows from CSV. Its score is an example calculation, not a domain-standard risk formula.

import { Asset, FailureMode } from "./libraries/asset-management.amx"
import { Screening } from "./libraries/screening.amx"

input asset: Asset
input screenings: Screening[]
input reviewedAt: DateTime

type ScreeningResult {
  assetId: String
  score: Number
  note?: String? = null
}

type Summary {
  assetName: String
  reviewedAt: DateTime
  totalScore: Number
}

fn screeningScore(severity: Number, occurrence: Number): Number = severity * occurrence

let scores: Number[] = for screening in screenings {
  return screeningScore(screening.severity, screening.occurrence)
}
let totalScore: Number = sum(scores)
export let summary: Summary = Summary { assetName: asset.name, reviewedAt: reviewedAt, totalScore: totalScore }
export let resultRows: ScreeningResult[] = for screening in screenings {
  return ScreeningResult { assetId: screening.assetId, score: screeningScore(screening.severity, screening.occurrence), note: screening.note }
}
